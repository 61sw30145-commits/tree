    function selectShotCount(count) {

      targetShotCount = count;

      capturedImages = new Array(count).fill(null);

      goToStep(2);

    }


    function goToStep(step, isBack = false) {

      if (isCapturing) stopCapture();


      if (currentStep === 3 && step !== 3) {

        stopCamera();

      }


      document.querySelectorAll('[id^="step-"]').forEach(el => el.classList.add('hidden'));

      document.getElementById(`step-${step}`).classList.remove('hidden');


      if (!isBack && currentStep !== step) {

        stepHistory.push(step);

      }

      currentStep = step;


      const backBtn = document.getElementById('back-button-container');

      if (step === 1) {

        backBtn.classList.add('hidden');

      } else {

        backBtn.classList.remove('hidden');

      }


      if (step === 2) renderFrames();

      if (step === 3) {

        if (!capturedImages[0]) {

          startCameraAndCapture();

        } else {

          showRetakeUI();

        }

      }

      if (step === 4) renderFinalCanvas(false);

    }


    function goBack() {

      if (stepHistory.length > 1) {

        stepHistory.pop();

        const prevStep = stepHistory[stepHistory.length - 1];

        goToStep(prevStep, true);

      }

    }


    function stopCapture() {

      isCapturing = false;

      if (captureTimer) clearInterval(captureTimer);

      document.getElementById('countdown').classList.add('hidden');

    }


    function stopCamera() {

      if (mediaStream) {

        mediaStream.getTracks().forEach(track => track.stop());

        mediaStream = null;

      }

    }

    async function startCameraAndCapture() {

      try {

        if (!mediaStream) {

          mediaStream = await navigator.mediaDevices.getUserMedia({ 

            video: { width: { ideal: 1920 }, height: { ideal: 1080 }, facingMode: "user" }, 

            audio: false 

          });

          document.getElementById('webcam').srcObject = mediaStream;

        }


        document.getElementById('webcam-container').classList.remove('hidden');

        document.getElementById('retake-controls').classList.add('hidden');

        document.getElementById('camera-guide').classList.remove('hidden');


        setTimeout(() => {

          capturedImages = new Array(targetShotCount).fill(null);

          isCapturing = true;

          captureNext(0);

        }, 1000);

      } catch (err) {

        alert("ไม่สามารถเข้าถึงกล้องได้ กรุณาอนุญาตสิทธิ์การใช้กล้อง");

      }

    }


    function captureNext(index) {

      if (!isCapturing || index >= targetShotCount) {

        showRetakeUI();

        return;

      }


      let count = 3;

      const cdEl = document.getElementById('countdown');

      cdEl.classList.remove('hidden');

      cdEl.innerText = count;


      captureTimer = setInterval(() => {

        count--;

        if (count > 0) {

          cdEl.innerText = count;

        } else {

          clearInterval(captureTimer);

          cdEl.classList.add('hidden');


          takePhoto(index);


          setTimeout(() => {

            if (isCapturing) captureNext(index + 1);

          }, 1000);

        }

      }, 1000);

    }


    function takePhoto(index) {

      const video = document.getElementById('webcam');

      const canvas = document.createElement('canvas');


      const vWidth = video.videoWidth;

      const vHeight = video.videoHeight;


      canvas.width = 975;

      canvas.height = 750;


      const targetRatio = 975 / 750;

      const videoRatio = vWidth / vHeight;


      let cropWidth, cropHeight, startX, startY;


      if (videoRatio > targetRatio) {

        cropHeight = vHeight;

        cropWidth = vHeight * targetRatio;

        startX = (vWidth - cropWidth) / 2;

        startY = 0;

      } else {

        cropWidth = vWidth;

        cropHeight = vWidth / targetRatio;

        startX = 0;

        startY = (vHeight - cropHeight) / 2;

      }


      const ctx = canvas.getContext('2d');

      ctx.translate(canvas.width, 0);

      ctx.scale(-1, 1);

      ctx.drawImage(video, startX, startY, cropWidth, cropHeight, 0, 0, canvas.width, canvas.height);


      capturedImages[index] = canvas.toDataURL('image/png');

    }


    function retakeSingleShot(index) {

      isCapturing = true;

      document.getElementById('webcam-container').classList.remove('hidden');

      document.getElementById('retake-controls').classList.add('hidden');

      document.getElementById('camera-guide').classList.remove('hidden');


      let count = 3;

      const cdEl = document.getElementById('countdown');

      cdEl.classList.remove('hidden');

      cdEl.innerText = count;


      captureTimer = setInterval(() => {

        count--;

        if (count > 0) {

          cdEl.innerText = count;

        } else {

          clearInterval(captureTimer);

          cdEl.classList.add('hidden');


          takePhoto(index);

          showRetakeUI();

        }

      }, 1000);

    }


    async function showRetakeUI() {

      isCapturing = false;

      document.getElementById('camera-guide').classList.add('hidden');

      document.getElementById('webcam-container').classList.add('hidden');


      const previewCanvas = document.getElementById('preview-framed-canvas');

      const ctx = previewCanvas.getContext('2d');


      previewCanvas.width = 1050;

      previewCanvas.height = 3150;


      ctx.fillStyle = "#FFFFFF";

      ctx.fillRect(0, 0, previewCanvas.width, previewCanvas.height);


      const rawImages = await Promise.all(capturedImages.map(imgSrc => loadImage(imgSrc)));


      const frameWidth = 975;

      const frameHeight = 750;

      const marginX = 37.5;

      const topYs = [975, 1801.5];


      rawImages.forEach((img, i) => {

        if (img) {

          ctx.drawImage(img, marginX, topYs[i], frameWidth, frameHeight);

        }

      });


      if (selectedFrame) {

        const frameImg = await loadImage(selectedFrame);

        if (frameImg) ctx.drawImage(frameImg, 0, 0, previewCanvas.width, previewCanvas.height);

      }


      document.getElementById('retake-controls').classList.remove('hidden');

    }


    function loadImage(src) {

      return new Promise((resolve, reject) => {

        if (!src) return resolve(null);

        const img = new Image();

        img.onload = () => resolve(img);

        img.onerror = reject;

        img.src = src;

      });

    }


    function createDitheredImage(imgElement, scale = 1.0, contrast = 1.35) {

      const canvas
