  <script>

    let selectedFrame = null;

    let targetShots = 3;

    let capturedImages = [];

    let frames = [];


    async function loadConfig() {

      try {

        const res = await fetch('config.json');

        const config = await res.json();

        frames = config.frames || [];

      } catch (e) {

        frames = [

          { id: 1, name: "1 ช็อต - กรอบใหญ่ 1", file: "frame_1shot_1.png", shots: 1 },

          { id: 2, name: "1 ช็อต - กรอบใหญ่ 2", file: "frame_1shot_2.png", shots: 1 },

          { id: 3, name: "บัวหลวงใหญ่", file: "frame1.png", shots: 3 },

          { id: 4, name: "กรอบสีดำ", file: "frame2.png", shots: 3 },

          { id: 5, name: "กรอบสีฟ้า", file: "frame3.png", shots: 3 },

          { id: 6, name: "ใบเสร็จ", file: "frame4.png", shots: 3 } 

        ];

      }

    }

    loadConfig();


    function goToStep(step) {

      document.querySelectorAll('[id^="step-"]').forEach(el => el.classList.add('hidden'));

      document.getElementById(`step-${step}`).classList.remove('hidden');


      if (step === 2) renderFrames();

      if (step === 3) {

        updateCameraGuide();

        if (capturedImages.length === 0 || capturedImages.every(img => img === null)) {

          startCameraAndCapture();

        } else {

          showRetakeUI();

        }

      }

      if (step === 4) {

        renderFinalCanvas();

      }

    }


    function selectShots(count) {

      targetShots = count;

      capturedImages = new Array(count).fill(null);

      goToStep(2);

    }


    function renderFrames() {

      const titleEl = document.getElementById('frame-title');

      titleEl.innerText = `เลือกกรอบรูป (${targetShots} ช็อต)`;


      const filteredFrames = frames.filter(f => f.shots === targetShots);

      const container = document.getElementById('frame-list');

      

      if (filteredFrames.length === 0) {

        container.innerHTML = `<p class="col-span-2 text-stone-400 py-8">ไม่พบกรอบรูปสำหรับ ${targetShots} ช็อต</p>`;

        return;

      }


      container.innerHTML = filteredFrames.map(f => `

        <div onclick="selectFrame('${f.file}')" class="group border border-stone-200 p-3 rounded-xl cursor-pointer hover:border-stone-800 bg-white shadow-sm hover:shadow-md transition flex flex-col items-center space-y-2">

          <div class="w-full aspect-[8/23] bg-stone-100 rounded-lg overflow-hidden flex items-center justify-center p-1 border border-stone-100 relative">

            <img src="${f.file}" alt="${f.name}" class="w-full h-full object-contain group-hover:scale-105 transition duration-200 z-10" onerror="this.src='https://via.placeholder.com/150x431?text=No+Preview'">

          </div>

          <p class="text-sm font-medium text-stone-700 group-hover:text-stone-900">${f.name}</p>

        </div>

      `).join('');

    }


    function selectFrame(file) {

      selectedFrame = file;

      goToStep(3);

    }


    function updateCameraGuide() {

      const guideBox = document.getElementById('guide-box');

      if (targetShots === 1) {

        guideBox.className = "w-full aspect-[700/1600] border-2 border-dashed border-white/90 shadow-[0_0_0_9999px_rgba(0,0,0,0.4)] relative flex items-center justify-center";

      } else {

        guideBox.className = "w-full aspect-[700/550] border-2 border-dashed border-white/90 shadow-[0_0_0_9999px_rgba(0,0,0,0.4)] relative flex items-center justify-center";

      }

    }


    async function startCameraAndCapture() {

      try {

        const stream = await navigator.mediaDevices.getUserMedia({ 

          video: { width: { ideal: 1920 }, height: { ideal: 1080 } }, 

          audio: false 

        });

        document.getElementById('webcam').srcObject = stream;

        document.getElementById('retake-controls').classList.add('hidden');

        document.getElementById('camera-guide').classList.remove('hidden');

        

        setTimeout(() => {

          capturedImages = new Array(targetShots).fill(null);

          captureNext(0);

        }, 1000);

      } catch (err) {

        alert("ไม่สามารถเข้าถึงกล้องได้ กรุณ่อนุญาตสิทธิ์การใช้กล้อง");

      }

    }


    function captureNext(index) {

      if (index >= targetShots) {

        showRetakeUI();

        return;

      }


      document.getElementById('shot-preview-overlay').classList.add('hidden');


      let count = 3;

      const cdEl = document.getElementById('countdown');

      cdEl.classList.remove('hidden');

      cdEl.innerText = count;


      const timer = setInterval(() => {

        count--;

        if (count > 0) {

          cdEl.innerText = count;

        } else {

          clearInterval(timer);

          cdEl.classList.add('hidden');

          

          takePhoto(index);


          const previewImg = document.getElementById('shot-preview-overlay');

          previewImg.src = capturedImages[index];

          previewImg.classList.remove('hidden');


          setTimeout(() => {

            captureNext(index + 1);

          }, 3000);

        }

      }, 1000);

    }


    function takePhoto(index) {

      const video = document.getElementById('webcam');

      const canvas = document.createElement('canvas');

      

      const targetW = 700;

      const targetH = targetShots === 1 ? 1600 : 550;


      canvas.width = targetW;

      canvas.height = targetH;

      

      const vWidth = video.videoWidth;

      const vHeight = video.videoHeight;

      

      const cropWidth = vWidth;

      const cropHeight = vWidth * (targetH / targetW);

      

      const startX = 0;

      const startY = Math.max(0, (vHeight - cropHeight) / 2);

      

      const ctx = canvas.getContext('2d');

      ctx.translate(canvas.width, 0);

      ctx.scale(-1, 1);

      

      ctx.drawImage(video, startX, startY, cropWidth, cropHeight, 0, 0, canvas.width, canvas.height);

      capturedImages[index] = canvas.toDataURL('image/png');

    }


    function retakeSingleShot(index) {

      document.getElementById('shot-preview-overlay').classList.add('hidden');

      document.getElementById('retake-controls').classList.add('hidden');

      document.getElementById('camera-guide').classList.remove('hidden');

      

      let count = 3;

      const cdEl = document.getElementById('countdown');

      cdEl.classList.remove('hidden');

      cdEl.innerText = count;


      const timer = setInterval(() => {

        count--;

        if (count > 0) {

          cdEl.innerText = count;

        } else {

          clearInterval(timer);

          cdEl.classList.add('hidden');

          takePhoto(index);

          

          const previewImg = document.getElementById('shot-preview-overlay');

          previewImg.src = capturedImages[index];

          previewImg.classList.remove('hidden');


          setTimeout(() => {

            document.getElementById('shot-preview-overlay').classList.add('hidden');

            showRetakeUI();

          }, 3000);

        }

      }, 1000);

    }


    function showRetakeUI() {

      document.getElementById('shot-preview-overlay').classList.add('hidden');

      document.getElementById('camera-guide').classList.add('hidden');


      const previewContainer = document.getElementById('shots-preview');

      previewContainer.innerHTML = capturedImages.map((img, i) => `

        <div class="relative cursor-pointer border-2 border-stone-300 hover:border-stone-800 rounded overflow-hidden group" onclick="retakeSingleShot(${i})">

          <img src="${img}" class="${targetShots === 1 ? 'w-20 h-32' : 'w-20 h-16'} object-cover">

          <div class="absolute inset-0 bg-black/40 flex items-center justify-center opacity-90 transition">

            <span class="text-white text-[11px] font-medium bg-black/60 px-1.5 py-0.5 rounded">ถ่ายใหม่</span>

          </div>

        </div>

      `).join('');


      document.getElementById('retake-controls').classList.remove('hidden');

    }


    function renderFinalCanvas() {

      const canvas = document.getElementById('final-canvas');

      const ctx = canvas.getContext('2d');


      canvas.width = 800;

      canvas.height = 2300;


      ctx.fillStyle = "#FFFFFF";

      ctx.fillRect(0, 0, canvas.width, canvas.height);


      let loadedCount = 0;


      if (targetShots === 1) {

        const img = new Image();

        img.onload = () => {

          ctx.drawImage(img, 50, 150, 700, 1600);

          drawFrameOverlay(canvas, ctx);

        };

        img.src = capturedImages[0];

      } else {

        capturedImages.forEach((imgSrc, i) => {

          const img = new Image();

          img.onload = () => {

            ctx.drawImage(img, 50, 150 + (i * 600), 700, 550);

            loadedCount++;

            

            if (loadedCount === 3) {

              drawFrameOverlay(canvas, ctx);

            }

          };

          img.src = imgSrc;

        });

      }

    }


    function drawFrameOverlay(canvas, ctx) {

      if (selectedFrame) {

        const frameImg = new Image();

        frameImg.onload = () => {

          ctx.drawImage(frameImg, 0, 0, canvas.width, canvas.height);

        };

        frameImg.src = selectedFrame;

      }

    }

  </script>
